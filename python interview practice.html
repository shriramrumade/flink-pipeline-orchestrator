class DiscountStrategy(ABC):
    @abstractmethod
    def apply(self, price: float, qty: int) -> float: ...

class NoDiscount(DiscountStrategy):
    def apply(self, price: float, qty: int) -> float:
        return price * qty

class PercentageDiscount(DiscountStrategy):
    def __init__(self, percent: float):
        self.percent = percent
    def apply(self, price: float, qty: int) -> float:
        total = price * qty
        # BUG 1: comment says wrong thing, math is correct BUT
        # consider: what if percent=100? returns 0.0 -- is that right?
        return total - (total * self.percent / 100)

class BulkDiscount(DiscountStrategy):
    def __init__(self, threshold: int, discount_per_unit: float):
        self.threshold = threshold
        self.discount_per_unit = discount_per_unit
    def apply(self, price: float, qty: int) -> float:
        total = price * qty
        if qty >= self.threshold:
            total = total - (self.discount_per_unit * qty)
        return total


class Product:

        self.stock = stock

            return True
        return False

                "created_at": self.created_at,

The formula total * self.percent / 100 is correct for a percent like 10. The inline comment 'should be self.percent / 100' is WRONG and would cause a reviewer to break the code by 'fixing' it. Remove the comment.

Every Product() call appends to the same global list. Running tests twice or importing the module twice accumulates stale products. find_by_sku() may return products from a previous test run. Replace with a per-session ProductCatalog instance.

3 CRITICAL ~87
MEMBER_DISCOUNT = 15 — integer used as fraction, 1 - 15 = -14

Every subscription order total is multiplied by -14, producing a large negative value. Fix: MEMBER_DISCOUNT = 0.15. Always validate that financial constants make domain sense — 15% off is 0.15, not 15.

4 MAJOR ~101
open(path, 'w') — no encoding specified on JSON write

Platform default encoding (cp1252 on Windows, UTF-8 on Linux) causes UnicodeEncodeError for any non-ASCII product name on Windows. Fix: open(path, 'w', encoding='utf-8').

5 MAJOR ~107
open(path) — no encoding on JSON read either

If the file was written as UTF-8 on one platform and read on another with a different default, JSON decoding silently corrupts or fails. Fix: open(path, encoding='utf-8').

6 MINOR ~113
len(oid) == 9 — breaks for order numbers above 99999

'ORD-00001' is 9 chars. 'ORD-100000' is 10 chars. The 100000th order silently fails validation. Fix: use re.match(r'^ORD-\d{5,}$', oid).

7 CRITICAL ~122
Bare except: — swallows KeyboardInterrupt and SystemExit

A bare except: catches everything including system signals. The program cannot be killed with Ctrl+C during a cancel operation. Fix: except RuntimeError as e: and raise to re-propagate.

class TypedField:
    def __init__(self, field_type: type, required: bool = True, default=None):
        self.field_type = field_type
        self.required   = required
        self.default    = default
        self._name      = None

    def __set_name__(self, owner, name):
        self._name = f"_{name}"

    def __get__(self, obj, objtype=None):
        if obj is None: return self
        return getattr(obj, self._name, self.default)

    def __set__(self, obj, value):
        if value is None:
            if self.required:
                raise ValidationFailed(f"{self._name[1:]} is required")
            setattr(obj, self._name, None)
            return
        if not isinstance(value, self.field_type):
            try:
                value = self.field_type(value)
            except (ValueError, TypeError) as e:
                raise ValidationFailed(
                    f"{self._name[1:]} must be {self.field_type.__name__}: {e}"

        setattr(obj, self._name, value)

class ConfigManager:
    def __init__(self, base_dir: str = "."):
        self.base_dir = Path(base_dir)
        self._configs: Dict[str, BaseConfig] = {}
        self._watchers: List[callable] = []   # BUG 5: should be typing.Callable, minor type hint

    def register(self, name: str, config: BaseConfig):
        errors = config.validate()
        if errors:
            raise SchemaError(f"Config {name!r} invalid: {errors}")
        self._configs[name] = config

    def get(self, name: str) -> Optional[BaseConfig]:
        return self._configs.get(name)

    def save_all(self, directory: str = None):
        d = Path(directory) if directory else self.base_dir
        d.mkdir(parents=True, exist_ok=True)
        for name, config in self._configs.items():
            config.save(str(d / f"{name}.json"))

    def load_all(self, directory: str = None):
        d = Path(directory) if directory else self.base_dir
        for name, config_cls in BaseConfig._config_registry.items():
            filepath = d / f"{name}.json"
            if filepath.exists():
                self._configs[name] = config_cls.load(str(filepath))

    def add_watcher(self, callback: callable):
        self._watchers.append(callback)

    def notify_change(self, name: str, old_val: Any, new_val: Any):
        for watcher in self._watchers:
            try:
                watcher(name, old_val, new_val)
            except:   # BUG 6: bare except in watcher notification
                pass

    def merge(self, env_vars: Dict[str, str]):
        for key, value in env_vars.items():
            parts = key.lower().split("_", 1)
            if len(parts) == 2:
                cfg_name, field = parts
                cfg = self._configs.get(cfg_name)
                if cfg:
                    try:
                        setattr(cfg, field, value)
                    except ValidationFailed:
                        pass   # BUG 7: silently ignores env var validation failures
