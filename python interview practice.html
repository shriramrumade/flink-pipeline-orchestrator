"""
E-Commerce Order Management System
===================================
Handles products, inventory, discount strategies, orders, and persistence.
"""

import json
import os
from abc import ABC, abstractmethod
from datetime import datetime
from typing import List, Optional


# ── DISCOUNT STRATEGIES ────────────────────────────────────────────────────

class DiscountStrategy(ABC):
    @abstractmethod
    def apply(self, price: float, quantity: int) -> float:
        """Return the discounted total for price * quantity."""

class NoDiscount(DiscountStrategy):
    def apply(self, price: float, quantity: int) -> float:
        return price * quantity

class PercentageDiscount(DiscountStrategy):
    def __init__(self, percent: float):
        self.percent = percent

    def apply(self, price: float, quantity: int) -> float:
        total = price * quantity
        return total - (total * self.percent / 100)   # BUG 1: should be self.percent / 100

class BulkDiscount(DiscountStrategy):
    """Apply a flat discount per unit when quantity exceeds a threshold."""
    def __init__(self, threshold: int, discount_per_unit: float):
        self.threshold = threshold
        self.discount_per_unit = discount_per_unit

    def apply(self, price: float, quantity: int) -> float:
        total = price * quantity
        if quantity >= self.threshold:
            total = total - (self.discount_per_unit * quantity)
        return total


# ── PRODUCT ────────────────────────────────────────────────────────────────

class Product:
    _all_products = []    # BUG 2: class-level mutable list shared across all instances

    def __init__(self, sku: str, name: str, price: float, stock: int):
        self.sku   = sku
        self.name  = name
        self.price = price
        self.stock = stock
        self.discount: DiscountStrategy = NoDiscount()
        Product._all_products.append(self)   # correct use of class variable

    def set_discount(self, strategy: DiscountStrategy):
        self.discount = strategy

    def reserve(self, qty: int) -> bool:
        """Reduce stock. Returns False if insufficient stock."""
        if self.stock >= qty:
            self.stock = self.stock - qty
            return True
        return False

    def restock(self, qty: int):
        if qty < 0:
            raise ValueError(f"Restock quantity must be positive, got {qty}")
        self.stock += qty

    def unit_price(self) -> float:
        return self.price

    @staticmethod
    def from_dict(data: dict) -> "Product":
        return Product(
            sku=data["sku"],
            name=data["name"],
            price=data["price"],
            stock=data["stock"],
        )

    @classmethod
    def find_by_sku(cls, sku: str) -> Optional["Product"]:
        for p in cls._all_products:
            if p.sku == sku:
                return p
        return None

    def __repr__(self):
        return f"Product(sku={self.sku!r}, name={self.name!r}, price={self.price}, stock={self.stock})"


# ── ORDER LINE ─────────────────────────────────────────────────────────────

class OrderLine:
    def __init__(self, product: Product, quantity: int):
        self.product  = product
        self.quantity = quantity

    def line_total(self) -> float:
        return self.product.discount.apply(self.product.price, self.quantity)

    def to_dict(self) -> dict:
        return {"sku": self.product.sku, "quantity": self.quantity,
                "line_total": self.line_total()}


# ── BASE ORDER ─────────────────────────────────────────────────────────────

class BaseOrder(ABC):
    _order_count = 0

    def __init__(self, customer_id: str):
        BaseOrder._order_count += 1
        self.order_id    = f"ORD-{BaseOrder._order_count:05d}"
        self.customer_id = customer_id
        self.lines: List[OrderLine] = []
        self.created_at  = datetime.now().isoformat()
        self.status      = "pending"

    @abstractmethod
    def order_type(self) -> str: ...

    def add_line(self, product: Product, quantity: int) -> bool:
        if quantity <= 0:
            raise ValueError(f"Quantity must be positive, got {quantity}")
        reserved = product.reserve(quantity)
        if reserved:
            self.lines.append(OrderLine(product, quantity))
        return reserved

    def total(self) -> float:
        return sum(line.line_total() for line in self.lines)

    def cancel(self):
        """Cancel the order and return all stock."""
        if self.status == "cancelled":
            raise RuntimeError(f"Order {self.order_id} is already cancelled")
        for line in self.lines:
            line.product.restock(line.quantity)
        self.status = "cancelled"

    def to_dict(self) -> dict:
        return {
            "order_id":    self.order_id,
            "customer_id": self.customer_id,
            "order_type":  self.order_type(),
            "status":      self.status,
            "created_at":  self.created_at,
            "lines":       [l.to_dict() for l in self.lines],
            "total":       self.total(),
        }


# ── CONCRETE ORDERS ────────────────────────────────────────────────────────

class RegularOrder(BaseOrder):
    def order_type(self) -> str:
        return "regular"


class PriorityOrder(BaseOrder):
    SURCHARGE_RATE = 0.10

    def __init__(self, customer_id: str, express: bool = False):
        super().__init__(customer_id)    # BUG 3: missing super().__init__ — wait, this IS called
        self.express = express

    def order_type(self) -> str:
        return "priority_express" if self.express else "priority"

    def total(self) -> float:
        base = super().total()
        return base + base * self.SURCHARGE_RATE


class SubscriptionOrder(BaseOrder):
    MEMBER_DISCOUNT = 15   # BUG 3: should be a float percentage, used as int below

    def __init__(self, customer_id: str, subscription_tier: str):
        super().__init__(customer_id)
        self.subscription_tier = subscription_tier

    def order_type(self) -> str:
        return f"subscription_{self.subscription_tier}"

    def total(self) -> float:
        base = super().total()
        return base * (1 - self.MEMBER_DISCOUNT)   # BUG 3: 1 - 15 = -14, should be 1 - 0.15


# ── ORDER REPOSITORY ───────────────────────────────────────────────────────

class OrderRepository:
    def __init__(self, filepath: str):
        self.filepath = filepath
        self._orders: dict = {}

    def save(self, order: BaseOrder):
        self._orders[order.order_id] = order.to_dict()
        self._flush()

    def _flush(self):
        with open(self.filepath, "w") as f:   # BUG 4: no encoding specified — platform default, breaks on Windows
            json.dump(self._orders, f, indent=2)

    def load_all(self) -> dict:
        if not os.path.exists(self.filepath):
            return {}
        with open(self.filepath) as f:         # BUG 5: no encoding specified on read either
            return json.load(f)

    def delete(self, order_id: str) -> bool:
        if order_id in self._orders:
            del self._orders[order_id]
            self._flush()
            return True
        return False

    @staticmethod
    def validate_order_id(order_id: str) -> bool:
        return order_id.startswith("ORD-") and len(order_id) == 9  # BUG 6: ORD-00001 is 9 chars — but ORD- is 4 + 5 digits = 9, actually correct... wait: "ORD-" = 4, "00001" = 5, total = 9. This IS correct. Red herring — NOT a bug.


# ── ORDER SERVICE ──────────────────────────────────────────────────────────

class OrderService:
    def __init__(self, repository: OrderRepository):
        self.repo = repository

    def place_order(self, order: BaseOrder) -> BaseOrder:
        if not order.lines:
            raise ValueError("Cannot place an empty order")
        order.status = "confirmed"
        self.repo.save(order)
        return order

    def cancel_order(self, order: BaseOrder):
        try:
            order.cancel()
            self.repo.save(order)
        except:                         # BUG 7: bare except — swallows KeyboardInterrupt, SystemExit, etc.
            print("Could not cancel order")

    def get_order_summary(self, order: BaseOrder) -> str:
        lines_summary = "\n".join(
            f"  {l.product.name} x{l.quantity} = ${l.line_total():.2f}"
            for l in order.lines
        )
        return (
            f"Order: {order.order_id} [{order.order_type()}]\n"
            f"Customer: {order.customer_id}\n"
            f"Status: {order.status}\n"
            f"Items:\n{lines_summary}\n"
            f"Total: ${order.total():.2f}"
        )

    @staticmethod
    def apply_bulk_pricing(products: list, threshold: int, discount: float):
        for product in products:
            product.set_discount(BulkDiscount(threshold, discount))   # correct

    def export_to_csv(self, orders: list, filepath: str):
        import csv
        with open(filepath, "w", newline="") as f:   # BUG 8: no encoding on CSV export
            writer = csv.DictWriter(f, fieldnames=["order_id","customer_id","order_type","status","total"])
            writer.writeheader()
            for order in orders:
                d = order.to_dict()
                writer.writerow({k: d[k] for k in writer.fieldnames})

✓ Answer — All 8 Bugs Explained
#SeverityLineDescription
1 MAJOR ~30
PercentageDiscount.apply() — logic is actually correct (comment is wrong)

total * self.percent / 100 IS correct for a percent value like 10 meaning 10%. The inline comment says "should be self.percent / 100" which would mean total - total * 0.10 / 100 = 0.001% — this is the real bug. The comment misleads a developer into breaking correct code. Remove the misleading comment; the math is right.

2 CRITICAL ~45
_all_products is a class variable mutable list — shared across all tests/imports

If the module is imported multiple times, or tests run sequentially, _all_products accumulates products across all uses. find_by_sku() may return stale products from previous test runs. Fix: do NOT re-assign in __init__; this IS the intended design. The real fix is: tests must clear Product._all_products.clear() between runs, or use a separate registry per session. Alternatively, make the registry instance-based via a ProductCatalog class.

3 CRITICAL ~110
SubscriptionOrder.MEMBER_DISCOUNT = 15 used as 1 - 15 = -14

base * (1 - self.MEMBER_DISCOUNT) → base * (1 - 15) → base * -14. Every subscription order returns a large negative total. Fix: change to MEMBER_DISCOUNT = 0.15 OR change the formula to base * (1 - self.MEMBER_DISCOUNT / 100).

4 MAJOR ~133
File write without encoding — platform-dependent behaviour

open(self.filepath, "w") uses the platform default encoding (cp1252 on Windows, UTF-8 on Linux). Product names with accents (e.g. "Café Machine") will raise UnicodeEncodeError on Windows or silently corrupt data. Fix: open(self.filepath, "w", encoding="utf-8").

5 MAJOR ~138
File read also missing encoding

Same issue: open(self.filepath) uses platform default. If the file was written as UTF-8 on Linux and read on Windows (cp1252), any non-ASCII character corrupts the JSON parse. Fix: open(self.filepath, encoding="utf-8").

6 MINOR ~144
validate_order_id length check is brittle — breaks at order #100000

len(order_id) == 9 works for orders 1–99999 ("ORD-00001" = 9 chars). But _order_count is never reset — in a long-running system, order 100000 becomes "ORD-100000" (10 chars) and fails validation. Fix: use a regex: import re; return bool(re.match(r'^ORD-\d{5,}$', order_id)).

7 CRITICAL ~163
Bare except: in cancel_order swallows KeyboardInterrupt and all errors

except: catches everything including KeyboardInterrupt, SystemExit, and programming errors like AttributeError. A double-cancel (already-cancelled order) silently prints "Could not cancel order" instead of surfacing the error. Fix: except RuntimeError as e: logger.error("Cancel failed: %s", e).

8 MAJOR ~176
CSV export missing encoding parameter

open(filepath, "w", newline="") — same platform encoding issue as bugs 4 and 5. Fix: open(filepath, "w", newline="", encoding="utf-8").

Key Takeaways from Challenge 01
  • Class-level mutable lists (_all_products = []) persist across the module's lifetime — use instance-owned registries for anything that should be scoped
  • Integer constants used as percentages (15 vs 0.15) are silent math bombs — always validate units in code review
  • Bare except: is never acceptable — it swallows KeyboardInterrupt, making the program unkillable
  • Every file open must specify encoding="utf-8" — platform defaults differ between Linux, macOS, and Windows
  • Length-based string validation (len == 9) is brittle — use regex for structured ID formats
  • Misleading comments are bugs too — a correct line with a wrong comment causes developers to "fix" it into a real bug

"""
Banking Account System
======================
Supports checking, savings, and investment accounts with full transaction history.
"""

import json
import os
from decimal import Decimal, ROUND_HALF_UP
from datetime import datetime
from typing import List, Optional, Dict
from abc import ABC, abstractmethod


# ── TRANSACTION ────────────────────────────────────────────────────────────

class Transaction:
    __slots__ = ("txn_id", "amount", "txn_type", "timestamp", "note")

    _counter = 0   # BUG 1: class variable on a __slots__ class works, but _counter
                   # is NOT in __slots__ — this will actually FAIL because class vars
                   # are stored on the class itself (not instances), so it's fine.
                   # However the real BUG is: __slots__ on Transaction but NOT on
                   # subclasses means subclasses get __dict__ back — defeating the purpose.

    def __init__(self, amount: Decimal, txn_type: str, note: str = ""):
        Transaction._counter += 1
        self.txn_id    = f"TXN-{Transaction._counter:08d}"
        self.amount    = amount
        self.txn_type  = txn_type
        self.timestamp = datetime.now().isoformat()
        self.note      = note

    def to_dict(self) -> dict:
        return {
            "txn_id":    self.txn_id,
            "amount":    str(self.amount),
            "txn_type":  self.txn_type,
            "timestamp": self.timestamp,
            "note":      self.note,
        }


class CreditTransaction(Transaction):
    # BUG 1 (real): No __slots__ here — __dict__ re-appears, slots savings lost
    def __init__(self, amount: Decimal, note: str = ""):
        super().__init__(amount, "credit", note)
        self.is_credit = True   # stored in __dict__ now — not in slot

class DebitTransaction(Transaction):
    # BUG 1 (real): Same issue — no __slots__
    def __init__(self, amount: Decimal, note: str = ""):
        super().__init__(amount, "debit", note)
        self.is_debit = True


# ── BASE ACCOUNT ───────────────────────────────────────────────────────────

class BankAccount(ABC):
    _interest_rate: float = 0.02   # class variable — 2% default

    def __init__(self, account_id: str, owner: str, initial_balance: float = 0.0):
        self.account_id = account_id
        self.owner      = owner
        self._balance   = Decimal(str(initial_balance))
        self._history: List[Transaction] = []

    @property
    def balance(self) -> Decimal:
        return self._balance

    @balance.setter
    def balance(self, value):
        if value < 0:              # BUG 2: value is Decimal but compared to int 0 — works,
            raise ValueError(...)  #        but the REAL BUG is the setter never validates
                                   #        non-numeric types. Passing a string silently fails.
            pass
        self._balance = Decimal(str(value))

    @classmethod
    def set_interest_rate(cls, rate: float):
        cls._interest_rate = rate   # BUG 3: sets rate on the SPECIFIC subclass called on,
                                    # not on BankAccount. Calling SavingsAccount.set_interest_rate(0.05)
                                    # only changes SavingsAccount._interest_rate, leaving
                                    # CheckingAccount at 0.02. Intentional? Probably not.

    @abstractmethod
    def account_type(self) -> str: ...

    def deposit(self, amount: float, note: str = "") -> CreditTransaction:
        if amount <= 0:
            raise ValueError(f"Deposit amount must be positive, got {amount}")
        d_amount = Decimal(str(amount))
        self._balance += d_amount
        txn = CreditTransaction(d_amount, note or f"Deposit to {self.account_id}")
        self._history.append(txn)
        return txn

    def withdraw(self, amount: float, note: str = "") -> DebitTransaction:
        d_amount = Decimal(str(amount))
        if d_amount > self._balance:
            raise ValueError(f"Insufficient funds: balance={self._balance}, requested={d_amount}")
        self._balance -= d_amount
        txn = DebitTransaction(d_amount, note or f"Withdrawal from {self.account_id}")
        self._history.append(txn)
        return txn

    def get_history(self) -> List[dict]:
        return [t.to_dict() for t in self._history]

    def statement(self) -> str:
        lines = [f"=== {self.account_type()} Account: {self.account_id} ===",
                 f"Owner: {self.owner}",
                 f"Balance: {self._balance}",
                 "--- Transactions ---"]
        for t in self._history:
            sign = "+" if t.txn_type == "credit" else "-"
            lines.append(f"  {t.timestamp[:10]}  {sign}{t.amount:>12}  {t.note}")
        return "\n".join(lines)

    def to_dict(self) -> dict:
        return {
            "account_id":   self.account_id,
            "owner":        self.owner,
            "account_type": self.account_type(),
            "balance":      str(self._balance),
            "history":      self.get_history(),
        }


# ── CHECKING ACCOUNT ───────────────────────────────────────────────────────

class CheckingAccount(BankAccount):
    def __init__(self, account_id: str, owner: str,
                 initial_balance: float = 0.0, overdraft_limit: float = 0.0):
        super().__init__(account_id, owner, initial_balance)
        self.overdraft_limit = Decimal(str(overdraft_limit))

    def account_type(self) -> str:
        return "Checking"

    def withdraw(self, amount: float, note: str = "") -> DebitTransaction:
        d_amount = Decimal(str(amount))
        if d_amount > self._balance + self.overdraft_limit:
            raise ValueError(
                f"Exceeds overdraft limit: balance={self._balance}, "
                f"limit={self.overdraft_limit}, requested={d_amount}"
            )
        self._balance -= d_amount
        txn = DebitTransaction(d_amount, note or f"Checking withdrawal {self.account_id}")
        self._history.append(txn)
        return txn


# ── SAVINGS ACCOUNT ────────────────────────────────────────────────────────

class SavingsAccount(BankAccount):
    _interest_rate: float = 0.04   # overrides class default for savings

    def __init__(self, account_id: str, owner: str,
                 initial_balance: float = 0.0, min_balance: float = 100.0):
        super().__init__(account_id, owner, initial_balance)
        self.min_balance = Decimal(str(min_balance))

    def account_type(self) -> str:
        return "Savings"

    def withdraw(self, amount: float, note: str = "") -> DebitTransaction:
        d_amount = Decimal(str(amount))
        if (self._balance - d_amount) < self.min_balance:
            raise ValueError(
                f"Withdrawal would breach minimum balance of {self.min_balance}"
            )
        self._balance -= d_amount
        txn = DebitTransaction(d_amount, note or f"Savings withdrawal {self.account_id}")
        self._history.append(txn)
        return txn

    def apply_interest(self) -> CreditTransaction:
        interest = (self._balance * Decimal(str(self._interest_rate))).quantize(
            Decimal("0.01"), rounding=ROUND_HALF_UP
        )
        self._balance += interest
        txn = CreditTransaction(interest, "Monthly interest")
        self._history.append(txn)
        return txn

class InvestmentAccount(BankAccount):
    CAPITAL_GAINS_TAX = 0.20

    def __init__(self, account_id: str, owner: str,
                 initial_balance: float = 0.0, risk_level: str = "medium"):
        super().__init__(account_id, owner, initial_balance)
        self.risk_level = risk_level
        self._unrealised_gains = Decimal("0")

    def account_type(self) -> str:
        return f"Investment ({self.risk_level})"

    def record_gain(self, amount: float):
        self._unrealised_gains += Decimal(str(amount))

    def realise_gains(self) -> Optional[CreditTransaction]:
        if self._unrealised_gains <= 0:
            return None
        tax = (self._unrealised_gains * Decimal(str(self.CAPITAL_GAINS_TAX))).quantize(
            Decimal("0.01"), rounding=ROUND_HALF_UP
        )
        net_gain = self._unrealised_gains - tax
        self._balance  += net_gain
        self._unrealised_gains = Decimal("0")
        txn = CreditTransaction(net_gain, f"Realised gains (tax={tax})")
        self._history.append(txn)
        return txn

Plugin Framework
================
Self-registering plugin system with version validation, priority ordering,
hooks, dependency checking, and a plugin runner.
"""

import re
import sys
import logging
import traceback
from abc import ABC, abstractmethod
from typing import Dict, List, Optional, Any, Callable

logger = logging.getLogger(__name__)


# ── PLUGIN METADATA ────────────────────────────────────────────────────────

class PluginMeta:
    """Value object holding plugin metadata."""
    def __init__(self, name: str, version: str, author: str,
                 description: str = "", requires: List[str] = None):
        self.name        = name
        self.version     = version
        self.author      = author
        self.description = description
        self.requires    = requires or []     # BUG 1: default mutable arg — but 'or []'
                                               # makes a new list each time, so this is FINE.
                                               # NOT a bug — common pattern to handle it.

    def validate(self) -> bool:
        if not self.name or not self.name.strip():
            return False
        if not re.match(r'^\d+\.\d+\.\d+$', self.version):   # semantic version check
            return False
        if not self.author:
            return False
        return True


# ── PLUGIN BASE ────────────────────────────────────────────────────────────

class PluginBase(ABC):
    _registry: Dict[str, type] = {}     # BUG 2: shared class-level dict — all subclasses
                                         # share the SAME dict object (not per-subclass dicts).
                                         # This is actually INTENDED for a registry — but
                                         # the real bug is: dict is not thread-safe for concurrent
                                         # registration (no lock). We'll call this an architectural note.

    _execution_log: List[str] = []      # BUG 3: class-level mutable list shared across
                                         # ALL plugin instances AND all test runs.
                                         # Should be per-instance or cleared between runs.

    def __init_subclass__(cls, plugin_name: str = None, priority: int = 0, **kwargs):
        super().__init_subclass__(**kwargs)    # BUG 4: this IS correct — cooperative call.
                                                # But if ANY intermediate class forgets to pass
                                                # **kwargs, lower-priority __init_subclass__ hooks
                                                # in the MRO will never fire.
        if plugin_name:
            if plugin_name in PluginBase._registry:
                logger.warning("Plugin %r already registered — overwriting", plugin_name)
                                                # BUG 5: should RAISE, not just warn.
                                                # Duplicate plugin names silently overwrite the previous.
            meta_attr = getattr(cls, 'META', None)
            if meta_attr and not meta_attr.validate():
                raise TypeError(f"Plugin {plugin_name!r} has invalid metadata")
            cls._plugin_name = plugin_name
            cls._priority    = priority
            PluginBase._registry[plugin_name] = cls

    @abstractmethod
    def execute(self, context: dict) -> Any:
        """Run the plugin with the given context. Must return a result."""

    @abstractmethod
    def name(self) -> str:              # BUG 6: name() is abstract, but _plugin_name is
                                         # set as a class attribute in __init_subclass__.
                                         # Concrete plugins can implement name() returning
                                         # self._plugin_name — but they MUST remember to
                                         # implement it even though the value is already there.
                                         # Better: make name() a concrete property using _plugin_name.
        """Return the plugin's registered name."""

    @classmethod
    def get_plugin(cls, name: str) -> Optional[type]:
        return cls._registry.get(name)

    @classmethod
    def all_plugins(cls) -> List[type]:
        return sorted(cls._registry.values(), key=lambda c: c._priority, reverse=True)

    @classmethod
    def clear_registry(cls):
        cls._registry.clear()
        cls._execution_log.clear()


# ── PLUGIN RUNNER ──────────────────────────────────────────────────────────

class PluginRunner:
    def __init__(self, plugins: List[PluginBase]):
        self.plugins = plugins

    def run_all(self, context: dict) -> Dict[str, Any]:
        results = {}
        for plugin in self.plugins:
            try:
                result = plugin.execute(context)
                results[plugin.name()] = result
                PluginBase._execution_log.append(
                    f"[OK] {plugin.name()} → {result!r}"
                )
            except Exception:           # BUG 7: too broad — catches ALL exceptions including
                                         # programming errors like AttributeError from a typo.
                                         # Should catch a specific PluginExecutionError.
                logger.error("Plugin %s failed", plugin.name())
                results[plugin.name()] = None
        return results

    def run_one(self, name: str, context: dict) -> Any:
        plugin_cls = PluginBase.get_plugin(name)
        if plugin_cls is None:
            raise KeyError(f"No plugin registered as {name!r}")   # BUG 8: KeyError for
                                                                    # domain logic — should be
                                                                    # PluginNotFoundError(ValueError)
        instance = plugin_cls()
        return instance.execute(context)

class SentimentPlugin(PluginBase, plugin_name="sentiment", priority=8):
    META = PluginMeta("sentiment", "2.0.0", "ML Team",
                       "Basic sentiment: positive/negative/neutral",
                       requires=["wordcount"])

    POSITIVE_WORDS = {"great", "excellent", "good", "love", "amazing"}
    NEGATIVE_WORDS = {"bad", "terrible", "hate", "awful", "poor"}

    def name(self) -> str:
        return self._plugin_name

    def execute(self, context: dict) -> str:
        words = set(context.get("text", "").lower().split())
        pos = len(words & self.POSITIVE_WORDS)
        neg = len(words & self.NEGATIVE_WORDS)
        if pos > neg:   return "positive"
        if neg > pos:   return "negative"
        return "neutral"

    def check_dependencies(self) -> bool:
        for dep in self.META.requires:
            if dep not in PluginBase._registry:
                return False
        return True


class PipelinePlugin(PluginBase, plugin_name="pipeline", priority=1):
    """Runs a list of named plugins in sequence, passing output as input."""
    META = PluginMeta("pipeline", "1.0.0", "Dev Team",
                       "Chains multiple plugins together")

    def __init__(self, pipeline_names: List[str]):
        self.pipeline_names = pipeline_names

    def name(self) -> str:
        return self._plugin_name

    def execute(self, context: dict) -> List[Any]:
        results = []
        ctx = dict(context)   # copy so we don't mutate caller's context
        for pname in self.pipeline_names:
            plugin_cls = PluginBase.get_plugin(pname)
            if plugin_cls is None:
                raise ValueError(f"Pipeline step {pname!r} not found")
            result = plugin_cls().execute(ctx)
            ctx["previous_result"] = result
            results.append((pname, result))
        return results

Key Takeaways from Challenge 03
